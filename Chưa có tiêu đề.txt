local OBS_SIGNATURE = "HMOBS"
local ZW_ZERO = "\u{200B}"
local ZW_ONE = "\u{200C}"

local function EncodeSignal(Command, UserId, Extra)
    local SignalString = OBS_SIGNATURE .. "|" .. tostring(Command) .. "|" .. tostring(UserId)
    if Extra then
        local SanitizedExtra = tostring(Extra):gsub("|", "~")
        SignalString = SignalString .. "|" .. SanitizedExtra
    end
    local OutputTable = {}
    for Index = 1, #SignalString do
        local ByteChar = string.byte(SignalString, Index)
        for BitIndex = 7, 0, -1 do
            local BitValue = math.floor(ByteChar / 2 ^ BitIndex) % 2
            if BitValue == 1 then
                table.insert(OutputTable, ZW_ONE)
            else
                table.insert(OutputTable, ZW_ZERO)
            end
        end
    end
    return table.concat(OutputTable)
end

local function DecodeSignal(Text)
    if not Text then return nil end
    if #Text < 64 then return nil end
    local BitsTable = {}
    for Index = 1, #Text do
        local Char = string.sub(Text, Index, Index)
        if Char == ZW_ZERO then
            table.insert(BitsTable, "0")
        elseif Char == ZW_ONE then
            table.insert(BitsTable, "1")
        end
    end
    if #BitsTable < 8 then return nil end
    local OutputString = {}
    for Index = 1, #BitsTable - 7, 8 do
        local ByteValue = 0
        for BitIndex = 0, 7 do
            local BitString = BitsTable[Index + BitIndex]
            ByteValue = ByteValue * 2 + tonumber(BitString)
        end
        table.insert(OutputString, string.char(ByteValue))
    end
    local DecodedString = table.concat(OutputString)
    if string.sub(DecodedString, 1, #OBS_SIGNATURE) ~= OBS_SIGNATURE then
        return nil
    end
    local SplitParts = string.split(DecodedString, "|")
    if #SplitParts < 3 then
        return nil
    end
    return SplitParts[2], tonumber(SplitParts[3]), SplitParts[4]
end

        local MoveVector = (FlyCamCFrame.LookVector * MoveForward) + (FlyCamCFrame.RightVector * MoveSide) + (Vector3.new(0, 1, 0) * MoveUp)
        if MoveVector.Magnitude > 0 then
            FlyCamCFrame = FlyCamCFrame + MoveVector.Unit * CurrentSpeed * DeltaTime
        end
        
        if FlyCamSettings.MemePart and FlyCamSettings.MemePart.Parent then
            FlyCamSettings.MemePart.CFrame = FlyCamCFrame
            CurrentCamera.CFrame = FlyCamCFrame
        end
    end)

    FlyCamMouseConnection = UserInputService.InputChanged:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.MouseMovement then
            local Sensitivity = 0.003
            local YawRotation = CFrame.Angles(0, -Input.Delta.X * Sensitivity, 0)
            local PitchRotation = CFrame.Angles(-Input.Delta.Y * Sensitivity, 0, 0)
            FlyCamCFrame = FlyCamCFrame * YawRotation * PitchRotation
        end
    end)
end

local function StopFlyCam(ShouldTeleport)
    if not FlyCamSettings.Enabled then return end
    local FinalCFrame = FlyCamCFrame
    FlyCamSettings.Enabled = false
    if FlyCamMoveConnection then FlyCamMoveConnection:Disconnect() end
    if FlyCamMouseConnection then FlyCamMouseConnection:Disconnect() end
    if FlyCamSettings.MemePart then
        FlyCamSettings.MemePart:Destroy()
        FlyCamSettings.MemePart = nil
    end
    CurrentCamera.CameraType = Enum.CameraType.Custom
    pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.Default end)

    if ShouldTeleport then
        task.spawn(function()
            local Character = LocalPlayer.Character
            local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
            if not HRP then return end
            for _, Part in ipairs(Character:GetDescendants()) do
                if Part:IsA("BasePart") and Part.SetNetworkOwner then
                    pcall(function() Part:SetNetworkOwner(LocalPlayer) end)
                end
            end
            for i = 1, 20 do
                pcall(function()
                    HRP.CFrame = FinalCFrame
                    HRP.AssemblyLinearVelocity = Vector3.zero
                end)
                task.wait(0.05)
            end
            SendChatMessage("[KAS] Đã TP đến vị trí cam!")
        end)
    end
end

        end

local lastResetTimer = tick()

task.spawn(function()
    while true do
        task.wait(0.05)
        pcall(function()
            if State.M1Spam then
                local t = FindNearest(Config.AttackRange)
                if t then HitAll(t) end
            end
            if State.KASM1Block and CombatRemote then pcall(function() CombatRemote:FireServer("M1") end) end
            if State.KASFastCombo and CombatRemote then
                pcall(function()
                    CombatRemote:FireServer("M1")
                    CombatRemote:FireServer("M2")
                    CombatRemote:FireServer("Kick")
                    CombatRemote:FireServer("Block")
                    task.wait(0.02)
                    CombatRemote:FireServer("Block")
                end)
            end
            if State.GodMode then
                local c = LocalPlayer.Character
                if c then
                    local h = c:FindFirstChildOfClass("Humanoid")
                    if h then h.MaxHealth = math.huge h.Health = math.huge end
                end
            end
            if State.FixLagEffects then
                for _, o in ipairs(workspace:GetDescendants()) do
                    if o:IsA("ParticleEmitter") or o:IsA("Trail") or o:IsA("Smoke") or o:IsA("Fire") then o.Enabled = false end
                end
            end
            if State.Spin then
                local c = LocalPlayer.Character
                if c then local r = c:FindFirstChild("HumanoidRootPart") if r then r.CFrame = r.CFrame * CFrame.Angles(0, math.rad(15), 0) end end
            end
            if State.AntiStun then
                local c = LocalPlayer.Character
                if c then local h = c:FindFirstChildOfClass("Humanoid") if h then h.PlatformStand = false h.Sit = false h.AutoRotate = true end end
            end
            if State.AutoReset then
                if tick() - lastResetTimer >= Config.ResetTime then
                    if CustomResetEvent then pcall(function() CustomResetEvent:FireServer() end) end
                    lastResetTimer = tick()
                end
            end
            if State.ESP then
                for _, p in ipairs(PlayersService:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character and not p.Character:FindFirstChild("ESP_HL") then
                        local hl = Instance.new("Highlight", p.Character)
                        hl.Name = "ESP_HL"
                        hl.FillColor = Color3.fromRGB(255, 255, 255)
                        hl.OutlineColor = Color3.fromRGB(0, 0, 0)
                        hl.FillTransparency = 0.5
                    end
                end
            else
                for _, p in ipairs(PlayersService:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character then
                        local hl = p.Character:FindFirstChild("ESP_HL")
                        if hl then hl:Destroy() end
                    end
                end
            end
        end)
    end
end)
